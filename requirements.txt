# ─────────────────────────────────────────────────────────────────
#  INSTALACIÓN MANUAL (si no usas install.sh)
# ─────────────────────────────────────────────────────────────────
#
#  1. Dependencias del sistema:
#       sudo apt update
#       sudo apt install -y ffmpeg python3-venv python3-pip
#
#  2. Driver NVIDIA + CUDA runtime:
#       nvidia-smi                 # verificar driver >= 530 (para CUDA 12.1)
#       # Si no tienes driver: sudo apt install nvidia-driver-535 -y && sudo reboot
#
#  3. Entorno virtual:
#       python3 -m venv venv
#       source venv/bin/activate
#       pip install --upgrade pip
#
#  4. PyTorch con CUDA:
#       pip install torch --index-url https://download.pytorch.org/whl/cu121
#
#  5. Resto de dependencias:
#       pip install -r requirements.txt
#
#  6. Pre-descargar el modelo Qwen3-ASR-1.7B (opcional, ~3.4 GB):
#       python -c "from qwen_asr import Qwen3ASRModel; Qwen3ASRModel.from_pretrained('Qwen/Qwen3-ASR-1.7B', cache_dir='./models')"
#
#  7. Ejecutar manualmente:
#       python manager.py
#
#  8. Ejecutar como servicio 24x7 (systemd):
#       sudo cp systemd/transcriber.service /etc/systemd/system/
#       sudo cp systemd/alerts.service /etc/systemd/system/
#       sudo systemctl daemon-reload
#       sudo systemctl enable transcriber alerts
#       sudo systemctl start  transcriber alerts
#       sudo systemctl status transcriber
#       journalctl -u transcriber -f     # ver logs en vivo
#
# ─────────────────────────────────────────────────────────────────
#  MONITOREO
# ─────────────────────────────────────────────────────────────────
#
#    watch -n 2 nvidia-smi              # VRAM en vivo
#    htop                               # CPU por canal
#    tail -f logs/manager.log           # log del orquestador
#    tail -f logs/transcriber.log       # log de inferencia
#    tail -f output/canal_01_*.txt      # transcripción de un canal
#    python monitor.py                  # dashboard de transcripciones
#    python search.py "palabra"         # búsqueda full-text
#
# ─────────────────────────────────────────────────────────────────
#  ESTIMADOS DE RECURSOS (8 canales, Qwen3-ASR-1.7B fp16 en T1000)
# ─────────────────────────────────────────────────────────────────
#
#    VRAM:    ~5-7 GB (cabe en 8 GB)
#    RAM:     ~6-10 GB (cola + FFmpeg + workers + Python)
#    CPU:     ~60-80% en picos (FFmpeg decodifica 8 streams AAC)
#    Disco:   ~50-100 MB/día de transcripciones
#
# ─────────────────────────────────────────────────────────────────
#  AJUSTE FINO
# ─────────────────────────────────────────────────────────────────

#  Si la latencia aumenta (cola creciendo en logs):

#    → Reducir MAX_BATCH_SIZE en transcriber.py
#
#  Si hay OOM en GPU:
#    → Bajar MAX_BATCH_SIZE a 2 en transcriber.py
#    → Reducir MAX_CHANNELS en manager.py
